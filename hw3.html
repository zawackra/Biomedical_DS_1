healthcare_data = "https://raw.githubusercontent.com/jhu-advdatasci/2018/master/data/KFF/healthcare-spending.csv"

healthcare_df = pd.read_csv(healthcare_data, skiprows=2, skipfooter=16, engine='python',  index_col=0).drop(index=['United States', 'US'], errors='ignore')

df_transposed = healthcare_df.T
df_transposed.index = df_transposed.index.str.split('_').str[0]

df_transposed.plot(figsize=(12, 6))
plt.title("Healthcare Spending by State")
plt.xlabel("Year")
plt.ylabel("Spending ($)")
plt.legend(title="State", bbox_to_anchor=(1.02, 1), loc="upper left")
plt.tight_layout()
plt.show()

plt.grid(axis='x', linestyle='--', alpha=0.6)
